```typescript
// src/core/scheduler.ts
import { logger } from "../utils/logger";

export function scheduleJob(intervalMs: number, jobFn: () => Promise<void>) {
    return setInterval(async () => {
        try {
            await jobFn();
        } catch (error: any) {
            logger.error(`Scheduled job failed: ${error.message}`);
        }
    }, intervalMs);
}
```
